local Players = game:GetService("Players")
local VirtualInputManager = game:GetService("VirtualInputManager")
local RunService = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")
local GuiService = game:GetService("GuiService")
local player = Players.LocalPlayer

-- Ensure mobile controls work
player.DevTouchMovementMode = Enum.DevTouchMovementMode.Thumbstick
player.DevComputerMovementMode = Enum.DevComputerMovementMode.UserChoice

local screenGui = Instance.new("ScreenGui", player:WaitForChild("PlayerGui"))
screenGui.Name = "ParryToggleGui"
screenGui.ResetOnSpawn = false
screenGui.IgnoreGuiInset = true
screenGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling

local toggleButton = Instance.new("TextButton")
toggleButton.Size = UDim2.new(0, 140, 0, 50)
toggleButton.Position = UDim2.new(1, -160, 1, -70)
toggleButton.AnchorPoint = Vector2.new(0, 0)
toggleButton.BackgroundColor3 = Color3.fromRGB(0, 170, 255)
toggleButton.TextColor3 = Color3.new(1, 1, 1)
toggleButton.TextSize = 20
toggleButton.Text = "SGC Parry: OFF"
toggleButton.AutoButtonColor = true
toggleButton.Modal = false
toggleButton.Parent = screenGui

local speedLabel = Instance.new("TextLabel")
speedLabel.Size = UDim2.new(0, 200, 0, 40)
speedLabel.Position = UDim2.new(0, 10, 0, 10)
speedLabel.BackgroundTransparency = 0.5
speedLabel.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
speedLabel.TextColor3 = Color3.new(1, 1, 1)
speedLabel.TextScaled = true
speedLabel.Font = Enum.Font.SourceSansBold
speedLabel.Text = "Object Speed: N/A"
speedLabel.Parent = screenGui

local parryEnabled = false
toggleButton.MouseButton1Click:Connect(function()
    parryEnabled = not parryEnabled
    toggleButton.Text = parryEnabled and "SGC Parry: ON" or "SGC Parry: OFF"
    toggleButton.BackgroundColor3 = parryEnabled and Color3.fromRGB(0, 200, 0) or Color3.fromRGB(0, 170, 255)
end)

local dragging, dragStart, startPos
toggleButton.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        dragStart = input.Position
        startPos = toggleButton.Position
    end
end)

toggleButton.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = false
    end
end)

RunService.Heartbeat:Connect(function()
    if dragging then
        local delta = UserInputService:GetMouseLocation() - dragStart
        toggleButton.Position = UDim2.new(0, startPos.X.Offset + delta.X, 0, startPos.Y.Offset + delta.Y)
    end
end)

local function pressParry()
    VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.F, false, game)
    task.wait(0.05)
    VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.F, false, game)
end

local function isCircularObjectAndMovable(obj)
    if obj:IsA("BasePart") and obj.Shape == Enum.PartType.Ball and obj.Velocity.Magnitude > 0 then
        return true
    end
    if (obj:IsA("MeshPart") or obj:IsA("UnionOperation")) then
        local size = obj.Size
        if math.abs(size.X - size.Y) < 0.1 and math.abs(size.Y - size.Z) < 0.1 and obj.Velocity.Magnitude > 0 then
            return true
        end
    end
    return false
end

local function getClosestMovingCircularObject()
    local hrp = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local closest, minDist = nil, math.huge
    for _, obj in pairs(workspace:GetChildren()) do
        if isCircularObjectAndMovable(obj) then
            local dist = (hrp.Position - obj.Position).Magnitude
            if dist < minDist then
                closest = obj
                minDist = dist
            end
        end
    end
    return closest
end

local function getAdjustedParryTime(speed)
    local minTime, maxTime = 0.15, 0.25
    local minSpeed, maxSpeed = 50, 150
    local ratio = math.clamp((speed - minSpeed) / (maxSpeed - minSpeed), 0, 1)
    return maxTime - ((maxTime - minTime) * ratio)
end

local maxParryDistance = 10
local spamCooldown = 0.1
local lastParry = 0

-- TOUCH support (mobile)
UserInputService.TouchTap:Connect(function(touchPositions, gameProcessedEvent)
    if gameProcessedEvent or GuiService.MenuIsOpen then return end
    if parryEnabled and tick() - lastParry >= spamCooldown then
        pressParry()
        lastParry = tick()
    end
end)

-- KEYBOARD support
UserInputService.InputBegan:Connect(function(input, gameProcessedEvent)
    if gameProcessedEvent then return end
    if input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode == Enum.KeyCode.F then
        if parryEnabled and tick() - lastParry >= spamCooldown then
            pressParry()
            lastParry = tick()
        end
    end
end)

-- Main logic loop
RunService.Heartbeat:Connect(function()
    if not parryEnabled then return end

    local hrp = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
    local circle = getClosestMovingCircularObject()

    if circle and hrp then
        local distance = (hrp.Position - circle.Position).Magnitude
        local directionToPlayer = (hrp.Position - circle.Position).Unit
        local circleVelocity = circle.Velocity
        local speed = circleVelocity.Magnitude
        local approaching = circleVelocity:Dot(directionToPlayer) > 0
        local timeToHit = speed > 0 and (distance / speed) or math.huge

        speedLabel.Text = string.format("Object Speed: %.1f", speed)

        if approaching and speed >= 100 and distance <= maxParryDistance then
            if tick() - lastParry >= spamCooldown then
                pressParry()
                lastParry = tick()
            end
        elseif approaching then
            local parryTime = getAdjustedParryTime(speed)
            if timeToHit <= parryTime and tick() - lastParry >= spamCooldown then
                pressParry()
                lastParry = tick()
            end
        end
    else
        speedLabel.Text = "Object Speed: N/A"
    end
end)
